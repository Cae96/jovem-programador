
programa
{

  inclua biblioteca Util  // importando funcao util

	cadeia usuario,senha
	inteiro infinito =1
	cadeia bancoUsuario[] = { "andre","luis"}
	cadeia bancoSenha[] = {"1234","6789"}
	cadeia bancoUsuarioTipo[] = {"1","2"}

	funcao inicio()
	{
		inteiro idade
		inteiro tamanho = Util.numero_elementos(bancoUsuario), auxiliar = 0

		escreva(tamanho)
			
		faca
		{
			escreva("-------------- Sistema de Apoio ao Imigrante ------------\n")
      escreva("Usuario: ")
      leia(usuario)
      escreva("Senha: ")
      leia(senha)

			para (inteiro posicao = 0; posicao < tamanho; posicao++)
				{
				se (usuario ==	bancoUsuario[posicao])
				{auxiliar = posicao
				}
				senao
				{}
				}
		
			se (usuario == bancoUsuario[auxiliar] e senha == bancoSenha[auxiliar])
			{escreva("entrada OK")
			Util.aguarde(1000)
   
   // Se o usuario for um usuario do governo abrirá o menu_governo, caso o usuario seja um imigrante abrirá o menu_imigrante
   // Aqui deveria se conectar ao banco de dados
	
				se 
				(bancoUsuarioTipo[auxiliar] == "1")
				{        
					escreva("menu_governo")
					Util.aguarde(2000)
				}

				se 
				(bancoUsuarioTipo[auxiliar] == "2")
				{
					escreva("menu_imigrante")
					Util.aguarde(2000)
				}

			}

				senao{        
					escreva("Dados nao encontrados no sistema, verifique os dados e tente novamente")
					Util.aguarde(2000)
					limpa()
				}

			}
		enquanto (infinito < 2)
		
		escreva ("\nCorreto!\n")
	}
}
